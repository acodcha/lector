34/// @brief Returns whether a given character is the leading byte of a UTF-8 character. All UTF-8
35/// characters measure either one, two, three, or four bytes. UTF-8 characters that measure only one
36/// byte are the ASCII characters. UTF-8 character that measure two, three, or four bytes are
37/// multi-byte characters and consist of a leading byte with a specific binary pattern and one or
38/// more continuation bytes of the binary pattern 10xxxxxx.
39///
40/// 1. One-byte UTF-8 characters are the ASCII characters. Their first bit is 0 and their binary
41/// pattern is therefore 0xxxxxxx.
42///
43/// 2. Two-byte UTF-8 characters have a leading byte with the binary pattern 110xxxxx and one
44/// continuation byte with the binary pattern 10xxxxxx. Together, the two bytes therefore have
45/// the binary pattern 110xxxxx 10xxxxxx.
46///
47/// 3. Three-byte UTF-8 characters have a leading byte with the binary pattern 1110xxxx and two
48/// continuation bytes with the binary pattern 10xxxxxx. Together, the three bytes therefore have
49/// the binary pattern 1110xxxx 10xxxxxx 10xxxxxx.
50///
51/// 4. Four-byte UTF-8 characters have a leading byte with the binary pattern 11110xxx and three
52/// continuation bytes with the binary pattern 10xxxxxx. Together, the four bytes therefore have
53/// the binary pattern 11110xxx 10xxxxxx 10xxxxxx 10xxxxxx.
54/// @param[in] character The character to check.
55/// @return True if the character is a leading byte; false if the character is a continuation byte.
56[[nodiscard]] inline bool is_leading_byte(const char character) {
57 // Cast to an unsigned character to avoid undefined behavior with bitwise operations on signed
58 // characters. The binary pattern 10xxxxxx that identifies a continuation byte ranges from 0x80 to
59 // 0xBF in hexadecimal notation.
60 return (static_cast<unsigned char>(character) & 0xC0) != 0x80;
61}

63/// @brief Counts and returns the number of UTF-8 code points in a string of text. The number of
64/// UTF-8 code points is a useful approximation of the number of graphemes in the string, where
65/// ASCII characters and multi-byte UTF-8 characters are each counted as one unit of length.
66/// @param[in] text The string of text whose UTF-8 code points are to be counted.
67/// @return The number of UTF-8 code points in the string of text.
68[[nodiscard]] inline std::size_t code_points(const std::string_view text) {
69 std::size_t count{0UL};
70 for (const char character : text) {
71 if (lector::is_leading_byte(character)) {
72 ++count;
73 }
74 }
75 return count;
76}

78/// @brief Finds the exact byte [begin, end) index interval in a string of text where a specified
79/// code point resides.
80/// @param[in] text The string of text to parse.
81/// @param[in] code_point_index The index of the code point in the string of text.
82/// @return A pair that contains the begin and end byte indices of the specified code point. The end
83/// index is the classical C++ "one past the end" index. If the specified code point index is out of
84/// bounds, both returned indices are set to one past the end index of the string, which is the size
85/// of the string.
86[[nodiscard]] inline std::pair<std::size_t, std::size_t> byte_interval(
87 const std::string_view text, const std::size_t code_point_index) {
88 std::size_t current_code_point_index{0UL};
89 std::size_t begin_byte_index{text.size()};
90 for (std::size_t current_byte_index{0UL}; current_byte_index < text.size();
91 ++current_byte_index) {
92 if (lector::is_leading_byte(text.at(current_byte_index))) {
93 if (current_code_point_index == code_point_index) {
94 begin_byte_index = current_byte_index;
95 } else if (current_code_point_index == code_point_index + static_cast<std::size_t>(1UL)) {
96 // In this case, this is the start of the next code point, and therefore the end of the
97 // requested code point.
98 return std::pair<std::size_t, std::size_t>{begin_byte_index, current_byte_index};
99 }
100 ++current_code_point_index;
101 }
102 }
103 if (begin_byte_index < text.size()) {
104 // In this case, the requested code point is found, but it is the last code point in the string.
105 return std::pair<std::size_t, std::size_t>{begin_byte_index, text.size()};
106 }
107 // In this case, the requested code point index is out of bounds.
108 return std::pair<std::size_t, std::size_t>{text.size(), text.size()};
109}

111/// @brief Computes and returns the length of the longest word in a string of text. The length of a
112/// word is measured by its number of UTF-8 code points.
113/// @param[in] text The string of text whose longest word length is to be computed.
114/// @return The length of the longest word in the string of text.
115[[nodiscard]] inline std::size_t longest_word_length(const std::string_view text) {
116 std::size_t current_longest_word_length{0UL};
117 std::size_t index{0UL};
118 while (index < text.length()) {
119 // Skip over any whitespaces.
120 while (index < text.length() && std::isspace(static_cast<unsigned char>(text[index])) != 0) {
121 ++index;
122 }
123 // Return if the end of the string has been reached after skipping whitespaces.
124 if (index >= text.length()) {
125 break;
126 }
127 // The index now points to the start of the current word.
128 const std::size_t current_word_start{index};
129 // Find the end of the current word.
130 while (index < text.length() && std::isspace(static_cast<unsigned char>(text[index])) == 0) {
131 ++index;
132 }
133 // Obtain the current word.
134 const std::string_view current_word{
135 text.substr(current_word_start, index - current_word_start)};
136 // Compute the length of the current word.
137 const std::size_t current_word_length{lector::code_points(current_word)};
138 // Update the longest word length.
139 current_longest_word_length = std::max(current_longest_word_length, current_word_length);
140 }
141 return current_longest_word_length;
142}

144/// @brief Tokenizes a string of text into a vector of strings of text, where each string in the
145/// vector corresponds to a word in the original string. Words are defined as sequences of
146/// non-whitespace characters, and whitespace characters are used as delimiters. The function does
147/// not modify the original string and returns views into it, so the original string must remain
148/// valid for the lifetime of the returned vector.
149/// @param[in] text The string of text to be tokenized.
150/// @return A vector of strings of text, each corresponding to a word in the original string.
151[[nodiscard]] inline std::vector<std::string_view> tokenize(const std::string_view text) {
152 std::vector<std::string_view> words;
153 std::size_t begin_index{0UL};
154 while (begin_index < text.size()) {
155 while (begin_index < text.size()
156 && std::isspace(static_cast<unsigned char>(text[begin_index])) != 0) {
157 ++begin_index;
158 }
159 if (begin_index == text.size()) {
160 break;
161 }
162 std::size_t end_index{begin_index};
163 while (
164 end_index < text.size() && std::isspace(static_cast<unsigned char>(text[end_index])) == 0) {
165 ++end_index;
166 }
167 words.push_back(text.substr(begin_index, end_index - begin_index));
168 begin_index = end_index;
169 }
170 return words;
171}

173/// @brief Pads a string of text from the left with spaces to reach a specified length. If the
174/// string of text is longer than the specified length, it is unchanged.
175/// @param[in] text The string of text to pad from the left.
176/// @param[in] length The desired length of the padded string.
177/// @return The padded string of text.
178[[nodiscard]] inline std::string pad_left(const std::string_view text, const std::size_t length) {
179 const std::size_t text_length{lector::code_points(text)};
180 if (text_length >= length) {
181 return std::string(text);
182 }
183 return std::string(length - text_length, ' ') + std::string{text};
184}

199/// @brief Joins a vector of strings where each string corresponds to a line of text into a single
200/// string of text, with newline characters inserted between the lines, and the lines left-aligned.
201/// @param[in] lines Vector of strings to be joined and left-aligned.
202/// @return The joined and left-aligned string of text.
203[[nodiscard]] inline std::string join_and_left_align(const std::vector<std::string>& lines) {
204 // Handle the empty case immediately to prevent underflow later.
205 if (lines.empty()) {
206 return std::string{};
207 }
208 // Calculate the exact total size.
209 std::size_t total_size{0UL};
210 for (const std::string& line : lines) {
211 total_size += line.size();
212 }
213 // Add space for the newline separators (one less than the total number of lines).
214 total_size += lines.size() - static_cast<std::size_t>(1UL);
215 // Create and allocate the resulting text.
216 std::string text;
217 text.reserve(total_size);
218 // Append the first line.
219 text.append(lines.front());
220 // Append subsequent lines prefixed by a newline.
221 for (std::size_t line_index{1UL}; line_index < lines.size(); ++line_index) {
222 text.push_back('\n');
223 text.append(lines.at(line_index));
224 }
225 return text;
226}

228/// @brief Joins a vector of strings where each string corresponds to a line of text into a single
229/// string of text, with newline characters inserted between the lines, and the lines right-aligned.
230/// @param[in] lines Vector of strings to be joined and right-aligned.
231/// @return The joined and right-aligned string of text.
232[[nodiscard]] inline std::string join_and_right_align(const std::vector<std::string>& lines) {
233 // Handle the empty case immediately to prevent underflow later.
234 if (lines.empty()) {
235 return std::string{};
236 }
237 // Compute the line lengths and find the maximum line length.
238 std::vector<std::size_t> line_lengths;
239 line_lengths.reserve(lines.size());
240 std::size_t longest_line_length{0UL};
241 for (const std::string& line : lines) {
242 const std::size_t length{lector::code_points(line)};
243 line_lengths.push_back(length);
244 longest_line_length = std::max(length, longest_line_length);
245 }
246 // Compute the exact total byte size.
247 std::size_t total_size{0UL};
248 for (std::size_t line_index{0UL}; line_index < lines.size(); ++line_index) {
249 const std::size_t padding{longest_line_length - line_lengths.at(line_index)};
250 total_size += lines.at(line_index).size() + padding;
251 }
252 total_size += lines.size() - static_cast<std::size_t>(1UL);
253 // Create and allocate the resulting text.
254 std::string text;
255 text.reserve(total_size);
256 // Append lines with padding.
257 for (std::size_t line_index{0UL}; line_index < lines.size(); ++line_index) {
258 if (line_index > 0UL) {
259 text.push_back('\n');
260 }
261 const std::size_t padding{longest_line_length - line_lengths.at(line_index)};
262 text.append(padding, ' ');
263 text.append(lines.at(line_index));
264 }
265 return text;
266}

268/// @brief Joins a vector of strings where each string corresponds to a line of text into a single
269/// string of text, with newline characters inserted between the lines, and the lines
270/// centre-aligned. If the total required centre-aligning padding is odd, the text is biased by one
271/// space towards the left.
272/// @param[in] lines Vector of strings to be joined and centre-aligned.
273/// @return The joined and centre-aligned string of text.
274[[nodiscard]] inline std::string join_and_centre_align_with_left_bias(
275 const std::vector<std::string>& lines) {
276 // Handle the empty case immediately to prevent underflow later.
277 if (lines.empty()) {
278 return std::string{};
279 }
280 // Compute the line lengths and find the maximum line length.
281 std::vector<std::size_t> line_lengths;
282 line_lengths.reserve(lines.size());
283 std::size_t longest_line_length{0UL};
284 for (const std::string& line : lines) {
285 const std::size_t length{lector::code_points(line)};
286 line_lengths.push_back(length);
287 longest_line_length = std::max(length, longest_line_length);
288 }
289 // Compute the exact total byte size.
290 std::size_t total_size{0UL};
291 for (std::size_t line_index{0UL}; line_index < lines.size(); ++line_index) {
292 const std::size_t total_padding{longest_line_length - line_lengths.at(line_index)};
293 // Bias left. When the total number of padding spaces is odd, integer division rounds down,
294 // giving one less padding space to the left.
295 const std::size_t left_padding{total_padding / 2UL};
296 total_size += lines.at(line_index).size() + left_padding;
297 }
298 total_size += lines.size() - static_cast<std::size_t>(1UL);
299 // Create and allocate the resulting text.
300 std::string text;
301 text.reserve(total_size);
302 // Append lines with padding.
303 for (std::size_t line_index{0UL}; line_index < lines.size(); ++line_index) {
304 if (line_index > 0UL) {
305 text.push_back('\n');
306 }
307 const std::size_t total_padding{longest_line_length - line_lengths.at(line_index)};
308 const std::size_t left_padding{total_padding / 2UL};
309 text.append(left_padding, ' ');
310 text.append(lines.at(line_index));
311 }
312 return text;
313}

362/// @brief Wraps a string of text to a line length and returns the result as a sequence of strings
363/// of text where each string in the sequence represents one line of text.
364/// @param[in] text The string of text to wrap.
365/// @param[in] line_length The desired line length to use when wrapping. Must be strictly greater
366/// than zero. Very long words whose lengths exceed this line length are hyphenated.
367/// @return The resulting sequence of strings of text that contains one string per line.
368/// @throws std::invalid_argument if the desired line length is zero.
369[[nodiscard]] inline std::vector<std::string> wrap(
370 const std::string_view text, const std::size_t line_length) {
371 // Ensure the line length is valid.
372 if (line_length <= static_cast<std::size_t>(0UL)) {
373 throw std::invalid_argument("Invalid line length. Must be strictly greater than zero.");
374 }
375 // Tokenize the input string of text.
376 const std::vector<std::string_view> words{lector::tokenize(text)};
377 // Process the tokenized input string of text and assemble the wrapped lines.
378 std::vector<std::string> lines;
379 std::string current_line;
380 std::size_t current_line_code_point_size{0UL};
381 for (const std::string_view current_word : words) {
382 // Measure the current word.
383 const std::size_t current_word_code_point_size{lector::code_points(current_word)};
384 const std::size_t space_needed_for_hyphen{
385 (current_line_code_point_size > static_cast<std::size_t>(0UL)) ?
386 static_cast<std::size_t>(1UL) :
387 static_cast<std::size_t>(0UL)};
388 // Check if the current word fits on the current line.
389 if (current_line_code_point_size > static_cast<std::size_t>(0UL)
390 && current_line_code_point_size + current_word_code_point_size + space_needed_for_hyphen
391 <= line_length) {
392 // In this case, the current word fits on the current line.
393 current_line.push_back(' ');
394 current_line.append(current_word);
395 current_line_code_point_size += current_word_code_point_size + space_needed_for_hyphen;
396 } else {
397 // In this case, the current word does not fit on the current line and must be wrapped to the
398 // next line.
399 if (current_line_code_point_size > static_cast<std::size_t>(0UL)) {
400 lines.push_back(std::move(current_line));
401 current_line.clear();
402 current_line_code_point_size = static_cast<std::size_t>(0UL);
403 }
404 // Check if the current word needs to be hyphenated.
405 if (current_word_code_point_size <= line_length) {
406 // In this case, the current word fits completely on an empty line and does not need to be
407 // hyphenated.
408 current_line = current_word;
409 current_line_code_point_size = current_word_code_point_size;
410 } else {
411 // In this case, the current word is too long and must be hyphenated.
412 std::string_view remaining_word{current_word};
413 std::size_t remaining_code_point_size{current_word_code_point_size};
414 // Iterate until the remaining portion of the current word fits on a line, and repeat as
415 // necessary; a very long word might need to be hyphenated multiple times.
416 while (remaining_code_point_size > line_length) {
417 // If the line_length is 1, no hyphen is used. Otherwise, take "line length - 1" code
418 // points to save 1 character for the hyphen.
419 const std::size_t chunk_code_point_size{line_length == static_cast<std::size_t>(1UL) ?
420 static_cast<std::size_t>(1UL) :
421 line_length - static_cast<std::size_t>(1UL)};
422 const std::size_t split_byte_index{
423 lector::byte_interval(remaining_word, chunk_code_point_size).first};
424 std::string split_line(
425 remaining_word.substr(static_cast<std::size_t>(0UL), split_byte_index));
426 if (line_length > static_cast<std::size_t>(1UL)) {
427 split_line.push_back('-');
428 }
429 lines.push_back(std::move(split_line));
430 remaining_word = remaining_word.substr(split_byte_index);
431 remaining_code_point_size -= chunk_code_point_size;
432 }
433 // The remaining slice of the word seeds the subsequent line.
434 if (remaining_code_point_size > static_cast<std::size_t>(0UL)) {
435 current_line = remaining_word;
436 current_line_code_point_size = remaining_code_point_size;
437 }
438 }
439 }
440 }
441 // Push the final built line if it is not empty.
442 if (current_line_code_point_size > static_cast<std::size_t>(0UL)) {
443 lines.push_back(std::move(current_line));
444 }
445 // Return the wrapped lines.
446 return lines;
447}

471/// @brief Collates two strings of text, each representing a column, into a single string that
472/// contains newline-separated lines of text, with the lines formatted such that the two columns are
473/// left-aligned and spaced a short distance apart.
474/// @param[in] first_column_text The string of text for the first column.
475/// @param[in] first_column_width The desired width of the first column. Very long words whose
476/// length exceeds this width are hyphenated.
477/// @param[in] second_column_text The string of text for the second column.
478/// @param[in] second_column_width The desired width of the second column. Very long words whose
479/// length exceeds this width are hyphenated.
480/// @return The string that contains the collated text.
481/// @throws std::invalid_argument if either desired column width is zero.
482[[nodiscard]] inline std::string collate_and_left_align(
483 const std::string_view first_column_text, const std::size_t first_column_width,
484 const std::string_view second_column_text, const std::size_t second_column_width) {
485 // Use a gutter width of two spaces.
486 constexpr std::size_t gutter_width{2UL};
487 // Wrap and split both columns.
488 const std::vector<std::string> first_column{lector::wrap(first_column_text, first_column_width)};
489 const std::vector<std::string> second_column{
490 lector::wrap(second_column_text, second_column_width)};
491 // Determine the total number of rows required.
492 const std::size_t rows{std::max(first_column.size(), second_column.size())};
493 // Pre-allocate memory for the result. A safe and highly efficient upper bound is the byte size of
494 // both original input strings, plus the maximum possible padding spaces and newlines per row.
495 std::string result;
496 result.reserve(first_column_text.length() + second_column_text.length()
497 + (rows * (first_column_width + gutter_width + static_cast<std::size_t>(1UL))));
498 // Collate the rows line by line.
499 for (std::size_t row_index{0UL}; row_index < rows; ++row_index) {
500 // Append a newline character for every row after the first to separate them without leaving a
501 // trailing newline at the very end of the string.
502 if (row_index > static_cast<std::size_t>(0UL)) {
503 result.push_back('\n');
504 }
505 // Grab the string for the first column if it exists on this row; otherwise, use an empty
506 // string.
507 const std::string_view first_cell{
508 row_index < first_column.size() ? std::string_view{first_column.at(row_index)} :
509 std::string_view{}};
510 result.append(first_cell);

512 // Otherwise, if the second column is exhausted, skip this padding to avoid unnecessary trailing
513 // whitespace.

515 const std::size_t first_cell_length{lector::code_points(first_cell)};
516 const std::size_t padding{first_column_width + gutter_width - first_cell_length};

524/// @brief Collates two strings of text, each representing a column, into a single string that
525/// contains newline-separated lines of text, with the lines formatted such that the two columns are
526/// right-aligned and spaced a short distance apart.
527/// @param[in] first_column_text The string of text for the first column.
528/// @param[in] first_column_width The desired width of the first column. Very long words whose
529/// length exceeds this width are hyphenated.
530/// @param[in] second_column_text The string of text for the second column.
531/// @param[in] second_column_width The desired width of the second column. Very long words whose
532/// length exceeds this width are hyphenated.
533/// @return The string that contains the collated text.
534/// @throws std::invalid_argument if either desired column width is zero.
535[[nodiscard]] inline std::string collate_and_right_align(
536 const std::string_view first_column_text, const std::size_t first_column_width,
537 const std::string_view second_column_text, const std::size_t second_column_width) {
538 // Use a gutter width of two spaces.
539 constexpr std::size_t gutter_width{2UL};
540 // Wrap and split both columns.
541 const std::vector<std::string> first_column{lector::wrap(first_column_text, first_column_width)};
542 const std::vector<std::string> second_column{
543 lector::wrap(second_column_text, second_column_width)};
544 // Determine the total number of rows required.
545 const std::size_t rows{std::max(first_column.size(), second_column.size())};
546 // Pre-allocate memory for the result. A safe and highly efficient upper bound is the byte size of
547 // both original input strings, plus the maximum possible padding spaces and newlines per row.
548 std::string result;
549 result.reserve(first_column_text.length() + second_column_text.length()
550 + (rows
551 * (first_column_width + gutter_width + second_column_width
552 + static_cast<std::size_t>(1UL))));
553 // Collate the rows line by line.
554 for (std::size_t row_index{0UL}; row_index < rows; ++row_index) {
555 // Append a newline character for every row after the first to separate them without leaving a
556 // trailing newline at the very end of the string.
557 if (row_index > static_cast<std::size_t>(0UL)) {
558 result.push_back('\n');
559 }
560 // Grab the string for the first column if it exists on this row; otherwise, use an empty
561 // string.
562 const std::string_view first_cell{
563 row_index < first_column.size() ? std::string_view{first_column.at(row_index)} :
564 std::string_view{}};
565 const std::size_t first_cell_length{lector::code_points(first_cell)};
566 // Calculate the leading padding. The ternary operator protects against std::size_t underflow in
567 // the extremely unlikely event a cell exceeds the column width.
568 const std::size_t first_cell_padding{
569 first_column_width > first_cell_length ? first_column_width - first_cell_length :
570 static_cast<std::size_t>(0UL)};
571 // Right-align the first column by prepending the required padding.
572 result.append(first_cell_padding, ' ');
573 result.append(first_cell);
574 // If the second column has text on this row, append the gutter, pad the second column, and
575 // append it. Otherwise, if the second column is exhausted, skip this padding to avoid
576 // unnecessary trailing whitespace.
577 if (row_index < second_column.size()) {
578 const std::string_view second_cell{second_column.at(row_index)};
579 const std::size_t second_cell_length{lector::code_points(second_cell)};
580 const std::size_t second_cell_padding{
581 second_column_width > second_cell_length ? second_column_width - second_cell_length :
582 static_cast<std::size_t>(0UL)};
583 result.append(gutter_width, ' ');
584 result.append(second_cell_padding, ' ');
585 result.append(second_cell);
586 }
587 }
588 return result;
589}
